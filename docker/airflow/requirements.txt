cryptography<46

# salida del balanceo depende del desempate entre distancias del vecino más cercano, así que una
# versión distinta de scikit-learn cambia unas filas sintéticas del dataset de entrenamiento.
scikit-learn==1.9.1

joblib==1.6.0

# Estas NO se pueden alinear con el repo (pandas 3.0.5, numpy 2.5.3, scipy 1.18.1, geopandas
# 1.2.0). La imagen base trae apache-airflow-providers-google y -snowflake, las dos con
# `pandas>=2.1.2,<2.2` como dependencia OBLIGATORIA. De ahí cascadea todo: pandas 2.1 declara
# `numpy<2`, y scipy 1.18 y geopandas 1.2 piden `numpy>=2`. Subirlas exige desinstalar esos dos
# providers, que nuestro DAG no usa y que ningún paquete necesita salvo detrás de un `extra`.

shapely>=2.0.0
sodapy>=1.8.0
